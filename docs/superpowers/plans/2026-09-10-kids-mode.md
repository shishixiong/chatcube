- [ ] **Step 2: 写实现**

```typescript
/**
 * 家长门算术题 — 纯函数（无 ArkUI 依赖，hypium 可测）。
 *
 * 题目两类随机混合:
 * - 两位数 × 一位数(3-9): 积 ≤ 891
 * - 三位数 ± 两位数: 和 ≤ 1098; 差保证 > 0
 * 目标用户 5 岁左右幼儿算不出来; 答错即换新题, 防同一题穷举。
 */

export interface GateQuestion {
  text: string
  answer: number
}

const MUL_MIN_A: number = 11
const MUL_MAX_A: number = 99
const MUL_MIN_B: number = 3
const MUL_MAX_B: number = 9
const ADDSUB_MIN_A: number = 100
const ADDSUB_MAX_A: number = 999
const ADDSUB_MIN_B: number = 10
const ADDSUB_MAX_B: number = 99
const MAX_ANSWER_INPUT_LENGTH: number = 4

function randomInt(min: number, max: number): number {
  return min + Math.floor(Math.random() * (max - min + 1))
}

export function generateGateQuestion(): GateQuestion {
  const kind: number = randomInt(0, 2)
  if (kind === 0) {
    const a: number = randomInt(MUL_MIN_A, MUL_MAX_A)
    const b: number = randomInt(MUL_MIN_B, MUL_MAX_B)
    return { text: `${a} × ${b}`, answer: a * b } as GateQuestion
  }
  if (kind === 1) {
    const a: number = randomInt(ADDSUB_MIN_A, ADDSUB_MAX_A)
    const b: number = randomInt(ADDSUB_MIN_B, ADDSUB_MAX_B)
    return { text: `${a} + ${b}`, answer: a + b } as GateQuestion
  }
  // 减法: b ≤ min(99, a-1), a ≥ 100 → 差恒 > 0
  const a: number = randomInt(ADDSUB_MIN_A, ADDSUB_MAX_A)
  const b: number = randomInt(ADDSUB_MIN_B, Math.min(ADDSUB_MAX_B, a - 1))
  return { text: `${a} − ${b}`, answer: a - b } as GateQuestion
}

export function checkGateAnswer(question: GateQuestion, input: string): boolean {
  const trimmed: string = input.trim()
  if (trimmed === '' || trimmed.length > MAX_ANSWER_INPUT_LENGTH) {
    return false
  }
  for (let i = 0; i < trimmed.length; i++) {
    const code: number = trimmed.charCodeAt(i)
    if (code < 48 || code > 57) {
      return false
    }
  }
  return Number.parseInt(trimmed, 10) === question.answer
}
```

注意：减法用 Unicode 减号 `−`（U+2212），测试里的 `q.text.includes('−')` 必须用同一字符。

  /**
   * 按会话聚合累计星星数（儿童模式主屏会话卡片徽章用）。
   * - 复用 getStarEventsSince(0) client-side 聚合, 与 StarRewardService.getAllTimeStarTotal 的 MVP 策略一致

   *   → UI 层优雅降级为不显示徽章

  async getStarTotalsBySession(): Promise<Map<string, number>> {
    const totals: Map<string, number> = new Map()
    const events: StarEventRow[] = await this.getStarEventsSince(0)
    for (let i = 0; i < events.length; i++) {
      const sessionId: string = events[i].sessionId
      if (sessionId === '') {
        continue
      }
      const previous: number = totals.get(sessionId) ?? 0
      totals.set(sessionId, previous + events[i].stars)
    }
    return totals
  }

```typescript
import { GateQuestion, generateGateQuestion, checkGateAnswer } from '../../utils/ParentalGateMath'

const GATE_ACCENT: string = '#FF9F43'
const GATE_ERROR: string = '#FF6B57'
const SHAKE_OFFSETS: number[] = [10, -10, 6, -6, 0]
const SHAKE_STEP_MS: number = 60

/**
 * 家长门（算术题验证）sheet 内容。
 * 答对 → onPassed（宿主负责关 sheet + 置解锁态）; 答错 → 红框 + 抖动 + 换新题; ✕/拖拽 → onCancelled。
 * 无状态持久化: 关闭即清空, 与"草稿纸"语义一致。
 */
@ComponentV2
export struct ParentalGateSheet {
  @Event onPassed: () => void = () => {}
  @Event onCancelled: () => void = () => {}
  @Local currentQuestion: GateQuestion = { text: '', answer: 0 } as GateQuestion
  @Local inputText: string = ''
  @Local showError: boolean = false
  @Local shakeX: number = 0
  private shakeTimerIds: number[] = []

  aboutToAppear(): void {
    this.currentQuestion = generateGateQuestion()
  }

  aboutToDisappear(): void {
    this.clearShakeTimers()
  }

  private clearShakeTimers(): void {
    for (let i = 0; i < this.shakeTimerIds.length; i++) {
      clearTimeout(this.shakeTimerIds[i])
    }
    this.shakeTimerIds = []
  }

  private handleConfirm(): void {
    if (checkGateAnswer(this.currentQuestion, this.inputText)) {
      this.onPassed()
      return
    }
    this.showError = true
    this.inputText = ''
    this.currentQuestion = generateGateQuestion()
    this.playShake()
  }

  private playShake(): void {
    this.clearShakeTimers()
    for (let i = 0; i < SHAKE_OFFSETS.length; i++) {
      const offset: number = SHAKE_OFFSETS[i]
      const timerId: number = setTimeout((): void => {
        this.shakeX = offset
      }, i * SHAKE_STEP_MS)
      this.shakeTimerIds.push(timerId)
    }
  }

  build() {
    Column({ space: 16 }) {
      // 标题栏: 左占位保持标题居中, 右侧 ✕
      Row() {
        Column()
          .width(36)
          .height(36)

        Text('家长验证')
          .fontSize(17)
          .fontWeight(FontWeight.Bold)
          .fontColor($r('app.color.text_primary'))
          .layoutWeight(1)
          .textAlign(TextAlign.Center)

        Button() {
          SymbolGlyph($r('sys.symbol.xmark'))
            .fontSize(18)
            .fontColor([$r('app.color.text_secondary')])
        }
        .type(ButtonType.Circle)
        .width(36)
        .height(36)
        .backgroundColor($r('app.color.divider'))
        .onClick((): void => {
          this.onCancelled()
        })
      }
      .width('100%')
      .alignItems(VerticalAlign.Center)

      Text('请算出下面的题目，答对即可进入')
        .fontSize(13)
        .fontColor($r('app.color.text_secondary'))

      Text(this.currentQuestion.text)
        .fontSize(34)
        .fontWeight(FontWeight.Bold)
        .fontColor($r('app.color.text_primary'))
        .translate({ x: this.shakeX })
        .animation({ duration: SHAKE_STEP_MS, curve: Curve.EaseOut })

      TextInput({ text: this.inputText, placeholder: '输入答案' })
        .type(InputType.Number)
        .fontSize(22)
        .textAlign(TextAlign.Center)
        .height(56)
        .borderRadius(14)
        .borderWidth(this.showError ? 2 : 0)
        .borderColor(GATE_ERROR)
        .onChange((value: string): void => {
          this.inputText = value
          this.showError = false
        })
        .onSubmit((): void => {
          this.handleConfirm()
        })

      if (this.showError) {
        Text('答案不对，换一题啦')
          .fontSize(13)
          .fontColor(GATE_ERROR)
      }

      Button('确定')
        .width('100%')
        .height(50)
        .borderRadius(14)
        .backgroundColor(GATE_ACCENT)
        .fontColor(Color.White)
        .fontSize(17)
        .onClick((): void => {
          this.handleConfirm()
        })
    }
    .width('100%')
    .padding({ left: 24, right: 24, top: 16, bottom: 32 })
  }
}
```
